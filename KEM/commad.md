# Create a project folder and move into it
mkdir pqc-lab
cd pqc-lab

# Create a virtual environment (keeps this isolated from your other Python projects)
python3 -m venv venv

# Activate it
# On Mac/Linux:
source venv/bin/activate
# On Windows (PowerShell):
venv\Scripts\Activate.ps1
# On Windows (cmd):
venv\Scripts\activate.bat
